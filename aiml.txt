def find_blank(board):
  for i in range(3):
    for j in range(3):
      if board[i][j] == 0:
        return i,j

def generate_moves(board):
  x,y = find_blank(board)

  moves = [(-1,0),(1,0),(0,-1),(0,1)]
  successors = []

  for dx,dy in moves:
    nx,ny = x + dx, y + dy

    if 0 <= nx < 3 and 0 <= ny < 3:
      new_board = [row[:] for row in board]

      new_board[x][y], new_board[nx][ny] = new_board[nx][ny], new_board[x][y]

      successors.append(new_board)

  return successors

def dfs(graph, start, goal):
  stack = [[start]]
  visited = set()

  while stack:
    path = stack.pop()
    node = path[-1]

    if node == goal:
      return path

    if node not in visited:
      visited.add(node)

      for neighbor in reversed(graph[node]):
        new_path = list(path)
        new_path.append(neighbor)
        stack.append(new_path)

  return None

import math

points=[(1,5),(6,4),(5,2),(2,1)]

x,y= 1.0, 1.0
step= 0.1

def total_distance(x,y):
  return sum((x-px)**2+(y-py)**2 for px,py in points)
current=total_distance(x,y)
iteration =1

while True:
  print(f"Iteration {iteration}: ({x:.1f}, {y:.1f}) Distance = {current:.2f}")

  neighbors=[
      (x+ step,y),
      (x- step,y),
      (x, y+step),
      (x, y-step)
  ]

  best_x,best_y=x,y
  best_dist=current

  for nx,ny in neighbors:
    d= total_distance(nx,ny)
    if d<best_dist:
      best_x,best_y=nx,ny
      best_dist=d


  if best_dist >= current:
    break

  x,y = best_x,best_y
  current = best_dist
  iteration+=1

print("\nOptional Point:",(round(x,2),round(y,2)))
print("Minimum Distance:", round(current,2))

class HelloProblem(SearchProblem):
    def actions(self, state):
        if len(state) < len(GOAL):
            return list("abcdefghijklmnopqrstuvwxyz")
        else:
            return []

    def result(self, state, action):

        return state == GOAL

        wrong = sum(
            1
            for i in range(len(state))
            if state[i] != GOAL[i]
        )
        missing = len(GOAL) - len(state)

class QueenChessBoard:
    def __init__(self, size):
        self.size = size
        self.columns=[]

    def place_in_next_row(self,column):
      self.columns.append(column)

    def remove_in_current_row(self):
      return self.columns.pop()

    def is_this_column_safe_in_next_row(self,column):
      row =len(self.columns)

      for queen_column in self.columns:
        if column== queen_column:
          return False

      for queen_row,queen_column in enumerate(self.columns):
        if queen_column-queen_row==column-row:
          return False

      for queen_row,queen_column in enumerate(self.columns):
        if (self.size-queen_column)-queen_row==(self.size-column)-row:
          return False

      return True

    def display(self):
      for row in range(self.size):
        for column in range(self.size):
          if column==self.columns[row]:
            print("Q",end=" ")
          else:
            print("*",end=" ")
        print()

def solve_queen(size):
  board = QueenChessBoard(size)
  number_of_solutions = 0
  row = 0
  column = 0

  while True:
    while column < size:
      if board.is_this_column_safe_in_next_row(column):
        board.place_in_next_row(column)
        row += 1
        column = 0
        break
      else:
        column += 1
    if column == size or row == size:
      if row == size:
        board.display()
        print()
        number_of_solutions += 1
        board.remove_in_current_row()
        row -= 1

      try:
        prev_column = board.remove_in_current_row()
      except IndexError:
        break

      row -= 1
      column = prev_column + 1
  print("Number of solution:",number_of_solutions)
n=int(input("Enter the number of queens:"))

graph ={
    'A':['B','C'],
    'B':['A','C','D'],
    'C':['A','B','D'],
    'D':['B','C'],
}

colors=['Red','Green','Blue']

assignment={}

def is_safe(region,color):
  for neighbor in graph[region]:
    if neighbor in assignment and assignment[neighbor]==color:
      return False
  return True

def map_coloring():
  if len(assignment)==len(graph):
    return True

  for region in graph:
    if region not in assignment:
      for color in colors:
        if is_safe(region,color):
          assignment[region]=color
          if map_coloring():
            return True
          del assignment[region]
      return False

if map_coloring():
  print("Map coloring solution:")
  for region in assignment:
    print(region,"->",assignment[region])
else:
  print("No solution exists.")

b)Alpha-Beta Pruning

def alphabeta(depth, index, max_player, values, alpha, beta):

    # Leaf node
    if depth == 0:
        return values[index]

    if max_player:
        best = -1000

        for i in range(2):
            value = alphabeta(depth - 1,
                              index * 2 + i,
                              False, values, alpha, beta)

            best = max(best, value)
            alpha = max(alpha, best)

            if beta <= alpha:
                break

        return best

    else:
        best = 1000

        for i in range(2):
            value = alphabeta(depth - 1,
                              index * 2 + i,
                              True, values, alpha, beta)

            best = min(best, value)
            beta = min(beta, best)

            if beta <= alpha:
                break

        return best


# Leaf values

from collections import deque
def valid(m, c):
    return 0 <= m <= 3 and 0 <= c <= 3 and \
           (m == 0 or m >= c) and \
           (3-m == 0 or 3-m >= 3-c)

def solve():
    start = (3, 3, 0)
    goal = (0, 0, 1)
    moves = [(1,0),(2,0),(0,1),(0,2),(1,1)]

    q=deque([(start,[start])])
    seen={start}
    while q:
      (m,c,b),path=q.popleft()

      if(m,c,b)==goal:
        return path

      for dm,dc in moves:
        if b==0:
          s=(m-dm, c-dc, 1)
        else:
          s=(m+dm, c+dc, 0)

        if valid(s[0],s[1]) and s not in seen:
                seen.add(s)
                q.append((s, path+[s]))

path=solve()
for i, s in enumerate(path):
  print(i,s)
